    public static void getSubsets(int i, List<Integer> t, int[] a) {
        if (i == a.length) { // condition to stop
            ab.add(new ArrayList<>(t));
            return;
        }
        t.add(a[i]);   // include
        getSubsets(i + 1, t, a);
        t.remove(t.size() - 1);// backtrack
        getSubsets(i + 1, t, a);  // exclude
    }


    public static void maxsubsetsofsum() {
        int[] nums = {2, 3, 5, 6, 8};
        int target = 10;

        List<List<Integer>> subsets = new ArrayList<>();
        subsets.add(new ArrayList<>()); // start with empty subset

        // generate all subsets
        for (int num : nums) {
            int size = subsets.size();

            for (int i = 0; i < size; i++) {
                List<Integer> newSubset = new ArrayList<>(subsets.get(i));
                newSubset.add(num);
                subsets.add(newSubset);
            }
        }

        // filter subsets with sum = target
        List<List<Integer>> result = new ArrayList<>();

        for (List<Integer> subset : subsets) {
            int sum = 0;
            for (int num : subset) {
                sum += num;
            }

            if (sum == target) {
                result.add(subset);
            }
        }

        System.out.println(result);

    }

    public static void mergeIntervals() {
//        [1,3] [2,6] [8,10] [9,12]
        List<List<Integer>> d = new ArrayList<>();
        d.add(List.of(1, 3));
        d.add(List.of(2, 6));
        d.add(List.of(8, 10));
        d.add(List.of(9, 12));
        List<List<Integer>> result = new ArrayList<>();


        for (int i = 0; i < d.size(); i++) {
            if (i == 0) {
                result.add(d.get(i));
            } else {
                if (d.get(i).get(0) <= d.get(i - 1).get(1)) {
                    int minvalue = Math.min(d.get(i - 1).get(0), d.get(i - 1).get(1));
                    int maxValue = Math.max(d.get(i).get(0), d.get(i).get(1));
                    result.add(List.of(minvalue, maxValue));
                    result.remove(i - 1);
                } else {
                    result.add(d.get(i));

                }
            }
        }
        System.out.println(result);

    }


    public static void mergeIntervalss() {

        List<List<Integer>> d = new ArrayList<>();

        d.add(List.of(1, 3));
        d.add(List.of(2, 6));
        d.add(List.of(8, 10));
        d.add(List.of(9, 12));

        List<List<Integer>> result = new ArrayList<>();

        for (List<Integer> current : d) {

            // First interval
            if (result.isEmpty()) {
                result.add(new ArrayList<>(current));
                continue;
            }

            // Get last merged interval
            List<Integer> last = result.get(result.size() - 1);

            // Check overlap
            if (current.get(0) <= last.get(1)) {

                int newStart = Math.min(last.get(0), current.get(0));
                int newEnd = Math.max(last.get(1), current.get(1));

                last.set(0, newStart);
                last.set(1, newEnd);

            } else {

                result.add(new ArrayList<>(current));
            }
        }

        System.out.println(result);
    }

    public static int balancedRuns(int[] times) {

        int count = 0;

        for (int i = 0; i < times.length; i++) {

            int leftSum = 0;
            int rightSum = 0;

            // Calculate left sum
            for (int j = 0; j < i; j++) {
                leftSum += times[j];
            }

            // Calculate right sum
            for (int j = i + 1; j < times.length; j++) {
                rightSum += times[j];
            }

            if (leftSum == rightSum) {
                count++;
            }
        }

        return count;
    }

    public static int longestSubArray(int[] arr, int k) {

        int maxLength = 0;

        for (int i = 0; i < arr.length; i++) {

            int sum = 0;

            for (int j = i; j < arr.length; j++) {

                sum += arr[j];

                if (sum <= k) {
                    maxLength = Math.max(maxLength, j - i + 1);
                }
            }
        }

        return maxLength;
    }

    public static int maxSubArraySumOfKWindow(int[] a, int k) {

        if (k <= 0 || k > a.length) {
            throw new IllegalArgumentException("Invalid window size");
        }

        Map<Integer, List<Integer>> d = new LinkedHashMap<>();

        for (int i = 0; i <= a.length - k; i++) {

            int sum = 0;
            List<Integer> list = new ArrayList<>();

            for (int j = i; j < i + k; j++) {
                sum += a[j];
                list.add(a[j]);
            }

            d.put(sum, list);
        }

        return Collections.max(d.keySet());
    }

    public static String removeAdjDuplicates(String s) {

        Stack<Character> stack = new Stack<>();

        for (char ch : s.toCharArray()) {

            if (!stack.isEmpty() && stack.peek() == ch) {
                stack.pop();
            } else {
                stack.push(ch);
            }
        }

        StringBuilder result = new StringBuilder();

        for (char ch : stack) {
            result.append(ch);
        }

        return result.toString();
    }

    public static int[] nextGreaterElement(int[] arr) {

        int n = arr.length;

        int[] result = new int[n];

        Stack<Integer> stack = new Stack<>();

        for (int i = n - 1; i >= 0; i--) {

            // Remove elements that cannot be the answer
            while (!stack.isEmpty() && stack.peek() <= arr[i]) {
                stack.pop();
            }

            // Stack top is the next greater element
            if (!stack.isEmpty()) {
                result[i] = stack.peek();
            } else {
                result[i] = -1;
            }

            // Add current element
            stack.push(arr[i]);
        }

        return result;
    }

    public static int longestConsecutive(int[] arr) {

        if (arr.length == 0) {
            return 0;
        }

        Arrays.sort(arr);

        int currentLength = 1;
        int maxLength = 1;

        for (int i = 1; i < arr.length; i++) {

            if (arr[i] == arr[i - 1]) {
                continue; // duplicate

            } else if (arr[i] == arr[i - 1] + 1) {
                currentLength++;

            } else {
                currentLength = 1;
            }

            maxLength = Math.max(maxLength, currentLength);
        }

        return maxLength;
    }

    public static void isSorted() {
        int[] arr = {10, 20, 30, 40, 50};
        boolean isSorted = true;
        // Null or arrays with 0 or 1 element are always considered sorted
        if (arr == null || arr.length <= 1) {
            isSorted = true;
        }

        // Check if any element is greater than the next element
        for (int i = 0; i < arr.length - 1; i++) {
            if (arr[i] > arr[i + 1]) {
                isSorted = false; // Found an out-of-order element
            }
        }
        System.out.println(isSorted);
    }


    public static String largestNumber(int[] nums) {
        String[] arr = new String[nums.length];

        for (int i = 0; i < nums.length; i++) {
            arr[i] = String.valueOf(nums[i]);
        }

        Arrays.sort(arr, (a, b) -> (b + a).compareTo(a + b));

        // Edge case: if largest is "0"
        if (arr[0].equals("0")) return "0";

        StringBuilder result = new StringBuilder();
        for (String s : arr) {
            result.append(s);
        }

        return result.toString();
    }

    public static int characterReplacementBrute(String s, int k) {
        int n = s.length();
        int maxLen = 0;

        for (int i = 0; i < n; i++) {
            int[] count = new int[26];
            int maxFreq = 0;

            for (int j = i; j < n; j++) {
                count[s.charAt(j) - 'A']++;
                maxFreq = Math.max(maxFreq, count[s.charAt(j) - 'A']);

                int window = j - i + 1;
                if (window - maxFreq <= k) {
                    maxLen = Math.max(maxLen, window);
                }
            }
        }

        return maxLen;
    }

    public static int getFirstIndexOfSubstring(String haystack, String needle) {
        int index = 0;
        int length = Math.max(haystack.length(), needle.length());
        if (needle.length() == 0 || needle.length() > haystack.length()) {
            index = 0;
        }
        for (int i = 0; i < length - needle.length(); i++) {
            String windowString = haystack.substring(i, i + needle.length());
            if (windowString.equals(needle)) {
                index = i;
                break;
            }
        }
        return index;
    }

    public static String longestPalindromeExpand(String s) {
        if (s == null || s.length() < 1) return "";

        int start = 0, end = 0;

        for (int i = 0; i < s.length(); i++) {
            // Case 1: Odd length palindrome (e.g., "aba" centered at 'b')
            int len1 = expandAroundCenter(s, i, i);

            // Case 2: Even length palindrome (e.g., "abba" centered between 'b' and 'b')
            int len2 = expandAroundCenter(s, i, i + 1);

            int maxLen = Math.max(len1, len2);

            // Update starting and ending indices if a longer palindrome is found
            if (maxLen > end - start) {
                start = i - (maxLen - 1) / 2;
                end = i + maxLen / 2;
            }
        }

        return s.substring(start, end + 1);
    }

    private static int expandAroundCenter(String s, int left, int right) {
        // Expand outwards as long as characters match and indices stay in bounds
        while (left >= 0 && right < s.length() && s.charAt(left) == s.charAt(right)) {
            left--;
            right++;
        }
        // Length of palindrome found
        return right - left - 1;
    }

    public static String longestPallindromeSubstring(String s) {
        String longest = "";
        for (int i = 0; i < s.length(); i++) {
            for (int j = i; j < s.length(); j++) {
                String substring = s.substring(i, j + 1);
                if (isPallindrome(substring) && substring.length() > longest.length()) {
                    longest = substring;
                }
            }
        }
        return longest;
    }

    public static int binarySearch(int[] arr, int target) {
        int low = 0, high = arr.length - 1;

        while (low <= high) {
            int mid = low + (high - low) / 2;

            if (arr[mid] == target) {
                return mid;
            } else if (arr[mid] < target) {
                low = mid + 1;   // go right
            } else {
                high = mid - 1;  // go left
            }
        }

        return -1; // not found
    }


    public static boolean isSubsequence(String s, String t) {
        int i = 0, j = 0;

        while (i < s.length() && j < t.length()) {
            if (s.charAt(i) == t.charAt(j)) {
                i++;
            }
            j++;
        }

        return i == s.length();
    }

    public static int maxSubArray(int[] nums) {
        int max = nums[0], sum = 0;

        for (int num : nums) {
            sum += num;
            max = Math.max(max, sum);
            if (sum < 0) sum = 0;
        }
        return max;
    }

    public static int trap(int[] height) {
        int left = 0, right = height.length - 1;
        int leftMax = 0, rightMax = 0, water = 0;

        while (left < right) {
            if (height[left] < height[right]) {
                if (height[left] >= leftMax)
                    leftMax = height[left];
                else
                    water += leftMax - height[left];
                left++;
            } else {
                if (height[right] >= rightMax)
                    rightMax = height[right];
                else
                    water += rightMax - height[right];
                right--;
            }
        }
        return water;
    }

    }

            }

            arr[n - 1] = first;

            System.out.print(num + " ");
        }
    }

    public static List<List<Integer>> threeSum(int[] nums) {
        Arrays.sort(nums);
        List<List<Integer>> result = new ArrayList<>();

        for (int i = 0; i < nums.length - 2; i++) {

            if (i > 0 && nums[i] == nums[i - 1]) continue;

            int left = i + 1;
            int right = nums.length - 1;

            while (left < right) {
                int sum = nums[i] + nums[left] + nums[right];

                if (sum == 0) {
                    result.add(Arrays.asList(nums[i], nums[left], nums[right]));

                    while (left < right && nums[left] == nums[left + 1]) left++;
                    while (left < right && nums[right] == nums[right - 1]) right--;

                    left++;
                    right--;

    public static String longestSubstring(String s) {

        if (s == null || s.length() == 0) return "";

        HashSet<Character> set = new HashSet<>();
        int left = 0, maxLength = 0, startIndex = 0;

        for (int right = 0; right < s.length(); right++) {

            while (set.contains(s.charAt(right))) {
                set.remove(s.charAt(left));
                left++;
            }

            set.add(s.charAt(right));

            if (right - left + 1 > maxLength) {
                maxLength = right - left + 1;
                startIndex = left;
            }
        }

        return s.substring(startIndex, startIndex + maxLength);
    }

    static String longestCommonPrefix_(String[] strs) {
        if (strs == null || strs.length == 0)
            return "-1";

        // Sort the array of strings
        Arrays.sort(strs);

        // Get the first and last strings after sorting
        String first = strs[0];
        String last = strs[strs.length - 1];
        int minLength
                = Math.min(first.length(), last.length());

        int i = 0;
        // Find the common prefix between the first and last
        // strings
        while (i < minLength
                && first.charAt(i) == last.charAt(i)) {
            i++;
        }

        // Check if there's no common prefix
        if (i == 0)
            return "-1";

        // Return the common prefix
        return first.substring(0, i);
    }

                }

    static boolean isBalanced(String str) {
        // 'open parentheses' - { [ (
        // 'close parentheses' - } ]
        // We will use stack to monitor last 'open parentheses'
        Stack<Character> st = new Stack<>(); // stack is LIFO (Last In, First Out) data structure

        // loop over input string
        for (char ch : str.toCharArray()) {
            // if char is 'open parentheses' push to our stack
            if (ch == '{' || ch == '(' || ch == '[') {
                st.push(ch);
                // if char is not open it might be 'close parentheses'
            } else {
                if (st.isEmpty()) {
                    return false;
                }

                // get latest 'open parentheses'
                char latestOpenedPar = st.pop();

                // checking here if char was 'close parentheses' then latest 'open parentheses'
                // should be appropriate to close one.
                if (latestOpenedPar == '{' && ch != '}') { // this for curly braces
                    return false;
                } else if (latestOpenedPar == '(' && ch != ')') { // this for parentheses
                    return false;
                } else if (latestOpenedPar == '[' && ch != ']') { // for square braces
                    return false;
                }

                // note, there is no else becuase if no condition match then we can continue checking next chars
            }
        }
        // make sure stack is empty, if not it's not balanced(for last input in the example)
        return st.size() == 0;
    }

    public static List<String> findSchedules() {

        String pattern = "08??840";
        int workHours = 24;
        int dayHours = 4;

        List<String> results = new ArrayList<>();

        // Calculate the sum of currently known hours
        int knownSum = 0;
        for (char c : pattern.toCharArray()) {
            if (c != '?') {
                knownSum += c - '0';
            }
        }

        // Calculate remaining hours that need to be distributed
        int remainingHours = workHours - knownSum;

        // Convert string to character array for efficient in-place modifications
        char[] scheduleArr = pattern.toCharArray();

        // Start recursive backtracking from index 0
        backtrack(scheduleArr, 0, remainingHours, dayHours, results);

        return results;
    }

    private static void backtrack(char[] schedule, int index, int remainingHours, int dayHours, List<String> results) {
        // Base case: if we processed the entire length of the schedule
        if (index == schedule.length) {
            // If all hours are distributed perfectly, add to the results list
            if (remainingHours == 0) {
                results.add(new String(schedule));
            }
            return;
        }

        // Optimization: stop early if hours left fall below 0
        if (remainingHours < 0) {
            return;
        }

        // If the current day is already fixed, move directly to the next index
        if (schedule[index] != '?') {
            backtrack(schedule, index + 1, remainingHours, dayHours, results);
            return;
        }

        // Try placing digits from 0 to dayHours sequentially (ensures lexicographical order)
        for (int hours = 0; hours <= dayHours; hours++) {
            if (hours <= remainingHours) {
                schedule[index] = (char) (hours + '0'); // Place current choice

                // Recurse for the next position
                backtrack(schedule, index + 1, remainingHours - hours, dayHours, results);

                schedule[index] = '?'; // Backtrack step to reset the position
            }
        }
    }
